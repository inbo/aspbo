SELECT "year", GBIF_EEARGCode( 1000, decimalLatitude, decimalLongitude, COALESCE(coordinateUncertaintyInMeters, 1000) ) AS eeaCellCode, classKey, class, genusKey, genus, COUNT(*) AS occurrences, MIN(COALESCE(coordinateUncertaintyInMeters, 1000)) AS minCoordinateUncertaintyInMeters, MIN(GBIF_TemporalUncertainty(eventDate, null)) AS minTemporalUncertainty FROM occurrence WHERE occurrenceStatus = 'PRESENT' AND genusKey IN (QD4HF, 7Q3FW, CVG4P, CTTYH, CVD64, CVLVZ, N37, NA, WQP, 78CX, 62HCR, 646HS, 768S, 63NN8, 3FYP, 53RF, 62HCQ, 63SCK, 6N56, 646YY, Y32, Q4MT4, 62FS7, 6BS5, 4V44, QF536, 5BVR) AND countrycode = 'BE' AND "year" >= 1900 AND hasCoordinate = TRUE AND genusKey IS NOT NULL AND NOT ARRAY_CONTAINS(issue, 'ZERO_COORDINATE') AND NOT ARRAY_CONTAINS(issue, 'COORDINATE_OUT_OF_RANGE') AND NOT ARRAY_CONTAINS(issue, 'COORDINATE_INVALID') AND NOT ARRAY_CONTAINS(issue, 'COUNTRY_COORDINATE_MISMATCH') AND (LOWER(identificationVerificationStatus) NOT IN ( 'unverified', 'unvalidated', 'not validated', 'under validation', 'not able to validate', 'control could not be conclusive due to insufficient knowledge', 'uncertain', 'unconfirmed', 'unconfirmed - not reviewed', 'validation requested' ) OR identificationVerificationStatus IS NULL) AND COALESCE(coordinateUncertaintyInMeters, 1000) <= 10000 GROUP BY "year", eeaCellCode, classKey, class, genusKey, genus ORDER BY "year" DESC, eeaCellCode ASC, genusKey ASC;
